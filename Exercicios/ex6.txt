double nota1 = 7.00;
double nota2 = 6.50;
double nota3 = 8.50;
Console.WriteLine($"Nota1: {nota1}");
Console.WriteLine($"Nota2: {nota2}");
Console.WriteLine($"Nota3: {nota3}");
Console.WriteLine();

double media = (nota1 + nota2 + nota3) / 3;
int frequencia = 75;
bool aprovado = media >= 7 && frequencia >= 75;
Console.WriteLine($"Frequeência = {frequencia}%");
Console.WriteLine($"Média = {media:F2}");
Console.WriteLine($"Aprovado? {aprovado}.");
